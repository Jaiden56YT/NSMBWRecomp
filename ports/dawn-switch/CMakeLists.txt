cmake_minimum_required(VERSION 3.16)
project(dawnprobe CXX)
set(CMAKE_CXX_STANDARD 20)
set(D ${CMAKE_CURRENT_LIST_DIR}/..)
add_executable(dawnprobe probe.cpp)
target_include_directories(dawnprobe PRIVATE ${D}/dawn/include ${D}/build/gen/include ${DEVKITPRO}/portlibs/switch/include)
target_compile_definitions(dawnprobe PRIVATE EGL_NO_PLATFORM_SPECIFIC_TYPES)
target_link_libraries(dawnprobe ${D}/build/src/dawn/native/libwebgpu_dawn.a -L${DEVKITPRO}/portlibs/switch/lib EGL glapi drm_nouveau stdc++ nx m)
nx_generate_nacp(dawnprobe.nacp NAME "Dawn probe" AUTHOR "NSMBWRecomp" VERSION "0.1")
nx_create_nro(dawnprobe NACP dawnprobe.nacp)
